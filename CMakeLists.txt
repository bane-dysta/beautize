cmake_minimum_required(VERSION 3.21)
project(beautize VERSION 0.1.0 LANGUAGES C CXX Fortran)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BEAUTIZE_OPENMP "Enable GFN-FF OpenMP parallelism" ON)
option(BEAUTIZE_BUILD_TESTS "Build native unit and integration tests" ON)
# All sources are bundled. No FetchContent, Python, ASE, or banelib dependency.
set(WITH_TESTS OFF CACHE BOOL "GFN-FF test-drive tests" FORCE)
set(WITH_OpenMP ${BEAUTIZE_OPENMP} CACHE BOOL "GFN-FF OpenMP" FORCE)
set(build_exe OFF CACHE BOOL "GFN-FF standalone applications" FORCE)
set(PYTHON_BINDINGS OFF CACHE BOOL "No Python bindings" FORCE)
add_subdirectory(third_party/gfnff EXCLUDE_FROM_ALL)
add_library(beautize_core STATIC
  src/gjf.cpp src/constraints.cpp src/optimize.cpp src/gfnff.cpp src/bridge.F90)
target_include_directories(beautize_core PUBLIC include)
target_link_libraries(beautize_core PUBLIC gfnff-lib)
set_target_properties(beautize_core PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  target_compile_options(beautize_core PRIVATE $<$<COMPILE_LANGUAGE:Fortran>:-ffree-line-length-none>)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(beautize_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
add_executable(beautize src/main.cpp)
target_link_libraries(beautize PRIVATE beautize_core)
include(GNUInstallDirs)
install(TARGETS beautize RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES README.md THIRD_PARTY.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/beautize)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/beautize/docs)
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/beautize/examples)
install(FILES licenses/GPL-3.0.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/beautize/licenses)
install(FILES third_party/gfnff/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/beautize/licenses RENAME gfnff-LGPL-3.0.txt)
if(BEAUTIZE_BUILD_TESTS)
  enable_testing()
  add_executable(beautize_tests tests/test_core.cpp)
  target_link_libraries(beautize_tests PRIVATE beautize_core)
  add_test(NAME native_core COMMAND beautize_tests)
  add_executable(beautize_backend_tests tests/test_backend.cpp)
  target_link_libraries(beautize_backend_tests PRIVATE beautize_core)
  add_test(NAME gfnff_backend COMMAND beautize_backend_tests)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/backend-test")
  set_tests_properties(gfnff_backend PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/backend-test")
  add_test(NAME cli_integration COMMAND ${CMAKE_COMMAND}
    -DBEAUTIZE=$<TARGET_FILE:beautize> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}
    -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-test -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.cmake)
  set_tests_properties(native_core gfnff_backend cli_integration PROPERTIES
    ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
endif()
